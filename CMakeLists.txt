cmake_minimum_required(VERSION 3.24)
project(VoiceIsolatorOBS VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype requires 64-bit Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VOICE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shared-voice" CACHE PATH "Voice Isolator RTX 1.2.0 source")
set(OBS_SOURCE "" CACHE PATH "OBS source tree with libobs public headers")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "Windows x64 obs.lib")
set(OBS_CONFIG_DIR "" CACHE PATH "Directory containing generated obsconfig.h")
if(NOT EXISTS "${VOICE_SOURCE}/runtime/worker.py" OR NOT EXISTS "${OBS_SOURCE}/libobs/obs-module.h" OR NOT EXISTS "${OBS_IMPORT_LIBRARY}")
  message(FATAL_ERROR "Set VOICE_SOURCE, OBS_SOURCE, OBS_CONFIG_DIR and OBS_IMPORT_LIBRARY.")
endif()
add_compile_options(/utf-8 /W4 /wd4201)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(voice-rtx-engine SHARED src/engine.cpp "${VOICE_SOURCE}/src/bridge.cpp")
target_include_directories(voice-rtx-engine PRIVATE "${VOICE_SOURCE}/src")
target_compile_definitions(voice-rtx-engine PRIVATE VOICE_ISOLATOR_RTX)
target_link_libraries(voice-rtx-engine PRIVATE ole32 advapi32)
add_library(voice-isolator-rtx MODULE src/filter.cpp)
target_include_directories(voice-isolator-rtx PRIVATE "${OBS_SOURCE}/libobs" "${OBS_CONFIG_DIR}")
target_link_libraries(voice-isolator-rtx PRIVATE "${OBS_IMPORT_LIBRARY}")
set(pkg voice-isolator-rtx)
install(FILES LICENSE DESTINATION "${pkg}/data/licenses" RENAME OBS-module-GPL-2.0.txt)
install(TARGETS voice-isolator-rtx LIBRARY DESTINATION "${pkg}" RUNTIME DESTINATION "${pkg}")
install(TARGETS voice-rtx-engine RUNTIME DESTINATION "${pkg}/data/worker/x64")
install(FILES "${VOICE_SOURCE}/runtime/worker.py" "${VOICE_SOURCE}/runtime/frame_graphs.py"
  DESTINATION "${pkg}/data/worker/Resources")
install(FILES "${VOICE_SOURCE}/LICENSE" DESTINATION "${pkg}/data/licenses" RENAME VoiceIsolator-MIT.txt)
install(FILES "${VOICE_SOURCE}/licenses/Inworld-LICENSE.txt" "${VOICE_SOURCE}/licenses/Inworld-NOTICE.txt"
  DESTINATION "${pkg}/data/licenses")

option(BUILD_INSTALL_PROBE "Check installed Windows layouts against real OBS runtimes" OFF)
if(BUILD_INSTALL_PROBE)
  add_executable(install-probe qa/check_install.cpp)
  target_include_directories(install-probe PRIVATE "${OBS_SOURCE}/libobs" "${OBS_CONFIG_DIR}")
  target_link_libraries(install-probe PRIVATE "${OBS_IMPORT_LIBRARY}")
endif()
